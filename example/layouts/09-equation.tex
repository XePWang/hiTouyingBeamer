\begin{frame}{公式与符号说明}
  \hitcolumnratio{.5}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{\hitleftwidth}
      \[ f(x)=x^2,\qquad 0\leq x\leq1 \]
    \end{column}
    \begin{column}{\hitrightwidth}
      \subhead{符号与条件}
      $x$ 表示输入，$f(x)$ 表示函数值。
      \notebox{结论只针对给出的定义域。}
    \end{column}
  \end{columns}
  \insight{在定义域内，$f(x)$ 随 $x$ 单调递增。}
\end{frame}
